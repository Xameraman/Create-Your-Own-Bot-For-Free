Write the Backend (app.py)

This is the brain of your bot. Copy this complete template into app.py. You only need to change BOT_BASE and GITHUB_BASE.

app.py — complete starter template
import logging, os, httpx, time, re, json
from fastapi import FastAPI, Request
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from groq import Groq
from collections import defaultdict

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("bot")

app = FastAPI()
app.add_middleware(CORSMiddleware, allow_origins=["*"],
                   allow_methods=["*"], allow_headers=["*"])

# ── API keys from HF Secrets ──
groq_key         = os.environ.get("GROQ_API_KEY")
turnstile_secret = os.environ.get("TURNSTILE_SECRET_KEY")

if groq_key:
    logger.info(f"✅ GROQ_API_KEY loaded ({len(groq_key)} chars)")
if turnstile_secret:
    logger.info(f"✅ TURNSTILE_SECRET_KEY loaded")

client = Groq(api_key=groq_key)

# ── Settings ──
MAX_MSG     = 500
DAILY_LIMIT = 150
CACHE_TTL   = 600   # 10 min knowledge cache
COOLDOWN    = 3     # seconds between messages

# ── Models — tried in order, first working one wins ──
MODELS = [
    "llama-3.1-8b-instant",    # fast
    "llama-3.3-70b-versatile",  # smart
    "gemma-7b-it",              # fallback
]

# ── CHANGE THIS to your GitHub username and repo name ──
GITHUB_BASE = "https://raw.githubusercontent.com/YOUR_USERNAME/YOUR_REPO/main"

# ── State ──
knowledge_cache = {}
user_daily      = defaultdict(lambda: {"count": 0, "date": ""})
user_last_msg   = defaultdict(float)
verified_users  = set()  # captcha verified — cached server-side

# ────────────────────────────────────────────────
# YOUR BOT — CHANGE THIS SECTION
# Replace with info about your own site/community
# ────────────────────────────────────────────────
BOT_BASE = """
You are [YOUR BOT NAME] — the official AI assistant for [YOUR WEBSITE].

[YOUR WEBSITE] is about: [DESCRIBE YOUR SITE IN 2-3 SENTENCES]

Topics you help with:
- [Topic 1 — e.g. how to download files]
- [Topic 2 — e.g. how to join the community]
- [Topic 3 — e.g. rules and guidelines]

Important rules:
- [Rule 1]
- [Rule 2]

Contact / Discord: [YOUR LINK]

Respond in a friendly, clear, and helpful way.
Use bullet points for step-by-step instructions.
If you don't know something specific, say so and
point them to your Discord or contact link.
"""

async def fetch_knowledge(source: str, message: str = "") -> str:
    """Load a .txt knowledge file from GitHub with 10-minute cache."""
    now = time.time()
    if source in knowledge_cache:
        if now - knowledge_cache[source]["t"] < CACHE_TTL:
            content = knowledge_cache[source]["c"]
            logger.info(f"📦 Cache hit: {source}")
            return content
    url = f"{GITHUB_BASE}/{source}.txt"
    try:
        async with httpx.AsyncClient(timeout=10.0) as c:
            r = await c.get(url)
            if r.status_code == 200:
                text = r.text[:4000]
                knowledge_cache[source] = {"c": text, "t": now}
                logger.info(f"📡 Fetched {source}.txt ({len(text)} chars)")
                return text
    except Exception as e:
        logger.error(f"❌ Knowledge fetch error: {e}")
    return ""

async def verify_captcha(token: str, ip: str) -> bool:
    """Verify Cloudflare Turnstile token. Returns True if human."""
    if not turnstile_secret:
        return True  # No secret = disabled, allow everyone
    if not token:
        return False
    try:
        async with httpx.AsyncClient(timeout=10.0) as c:
            r = await c.post(
                "https://challenges.cloudflare.com/turnstile/v0/siteverify",
                data={"secret": turnstile_secret, "response": token, "remoteip": ip}
            )
            result = r.json()
            if result.get("success"):
                logger.info(f"✅ Captcha verified: {ip}")
                return True
            logger.warning(f"🤖 Captcha failed: {result.get('error-codes')}")
            return False
    except:
        return True  # On error allow through — don't block real users

def get_user_id(request: Request) -> str:
    token = request.headers.get("x-user-token", "")
    if token and len(token) > 8: return f"token:{token}"
    fwd = request.headers.get("X-Forwarded-For")
    if fwd: return f"ip:{fwd.split(',')[0].strip()}"
    return f"ip:{request.client.host}"

def get_user_ip(request: Request) -> str:
    fwd = request.headers.get("X-Forwarded-For")
    return fwd.split(",")[0].strip() if fwd else request.client.host

def check_daily(user_id: str) -> bool:
    today = time.strftime("%Y-%m-%d")
    d = user_daily[user_id]
    if d["date"] != today: d["date"] = today; d["count"] = 0
    d["count"] += 1
    return d["count"] <= DAILY_LIMIT

class ChatRequest(BaseModel):
    message:       str
    history:       list = []
    source:        str  = "general"
    captcha_token: str  = ""

@app.post("/chat")
async def chat(req: ChatRequest, request: Request):
    user_id = get_user_id(request)
    user_ip = get_user_ip(request)

    # ── Captcha: verify once, cache server-side ──
    # Tokens are single-use! Never re-verify the same token.
    # Once a user passes, store their ID and skip verification.
    if user_id not in verified_users:
        if not await verify_captcha(req.captcha_token, user_ip):
            return {"ok": False, "captcha": True,
                    "response": "⛔ Verification failed. Please verify again."}
        verified_users.add(user_id)
        logger.info(f"✅ User verified: {user_id[:30]}")

    # ── Basic checks ──
    if not req.message.strip():
        return {"ok": False, "response": "Please type a message."}
    if len(req.message) > MAX_MSG:
        return {"ok": False, "response": f"Message too long. Max {MAX_MSG} characters."}
    if not check_daily(user_id):
        return {"ok": False, "response": "Daily limit reached. Try again tomorrow."}
    now = time.time()
    if now - user_last_msg[user_id] < COOLDOWN:
        return {"ok": False, "response": "⏳ Sending too fast. Please wait a moment."}
    user_last_msg[user_id] = now

    # ── Load knowledge for this page ──
    knowledge = await fetch_knowledge(req.source, req.message)

    # ── Build system prompt ──
    system_prompt = BOT_BASE
    if knowledge:
        system_prompt += f"\n\nKNOWLEDGE BASE FOR THIS PAGE:\n{knowledge}"
    system_prompt += "\n\nBe helpful and concise. Use bullet points for steps."

    messages = [{"role": "system", "content": system_prompt}]
    for msg in req.history[-8:]:
        role = "user" if msg["role"] == "user" else "assistant"
        messages.append({"role": role, "content": msg["content"]})
    messages.append({"role": "user", "content": req.message})

    # ── Try models in order ──
    for model in MODELS:
        try:
            logger.info(f"🤖 Trying: {model}")
            resp = client.chat.completions.create(
                model=model, messages=messages,
                temperature=0.5, max_tokens=1024, timeout=15
            )
            reply = resp.choices[0].message.content
            logger.info(f"✅ Response from {model}")
            return {"ok": True, "response": reply}
        except Exception as e:
            err = str(e).lower()
            if any(x in err for x in ["429", "decommissioned", "not found", "unavailable"]):
                logger.warning(f"⚠️ {model} skipped: {str(e)[:60]}")
                continue
            logger.error(f"❌ {model}: {e}")
            raise

    return {"ok": False,
            "response": "I'm at capacity right now. Please try again in a moment."}

@app.get("/")
def root(): return {"status": "Bot running", "version": "1.0"}

@app.get("/status")
def status(): return {"ok": True, "locked": False, "unlock_in": 0}

Add Cloudflare Turnstile Captcha

Captcha verification stops bots from spamming your chat and draining your free API quota. Cloudflare Turnstile is the best free option — it works invisibly for most real users.

✅
Why Turnstile? Completely free, no credit card, works on any website including GitHub Pages, and most real users pass it without doing anything at all — no puzzle solving required.

Step 1 — Create your Turnstile widget on Cloudflare

  • 1
    Go to dash.cloudflare.com → sign in → left sidebar: Turnstile
  • 2
    Click Add site → Site name: anything → Domain: your GitHub Pages domain, e.g. yourusername.github.io
  • 3
    Widget type: Managed — this is invisible, no puzzle for users
  • 4
    Click Create → you get two keys: Site Key (public — goes in HTML) and Secret Key (private — goes in HF Secrets only)
🔐
Never put your Secret Key in HTML or JavaScript. It goes only in Hugging Face Secrets. The Site Key is safe to put in HTML — it's designed to be public.

Step 2 — Add Secret Key to Hugging Face

  • 1
    HF Space → Settings → Variables and secrets → New secret
  • 2
    Name: TURNSTILE_SECRET_KEY → paste your Secret Key → Save
  • 3
    Click Factory reboot to restart with the new secret
⚠️
Important: Cloudflare tokens are single-use. Each token can only be verified once. The backend in this guide caches verification server-side after the first successful check, so users don't need to reverify every message. This is already handled in the app.py template above — don't change that logic or users will get stuck in a verification loop.

JavaScript — full widget logic
<script>
(function(){
  // ← CHANGE THESE TWO LINES for each page:
  const API    = 'https://YOUR-HF-USERNAME-YOUR-SPACE-NAME.hf.space/chat';
  const SOURCE = 'general'; // knowledge file name without .txt

  // Persistent user token for rate limiting
  let token = localStorage.getItem('bot_token');
  if (!token) { token = crypto.randomUUID(); localStorage.setItem('bot_token', token); }
  const HEADERS = {'Content-Type': 'application/json', 'x-user-token': token};

  // Captcha — sessionStorage clears on page refresh (intentional)
  // This means users verify once per browser session, not every message
  let captchaVerified = sessionStorage.getItem('bot_captcha') === 'yes';
  let captchaToken    = null;

  // Called by Cloudflare when user passes captcha
  window.onCaptchaDone = function(t) {
    captchaToken = t;
    captchaVerified = true;
    sessionStorage.setItem('bot_captcha', 'yes');
    document.getElementById('captcha-wrap').style.display = 'none';
    document.getElementById('bot-input-row').style.display = 'flex';
    document.getElementById('bot-footer').textContent = 'Powered by AI';
    input.focus();
  };

  function showCaptchaUI() {
    const w = document.getElementById('captcha-wrap');
    const r = document.getElementById('bot-input-row');
    const f = document.getElementById('bot-footer');
    if (captchaVerified) {
      w.style.display = 'none'; r.style.display = 'flex';
    } else {
      w.style.display = 'block'; r.style.display = 'none';
      if (f) f.textContent = '🔒 Verify to start chatting';
    }
  }

  let history=[], isOpen=false, isLoading=false;
  const btn   = document.getElementById('bot-btn');
  const panel = document.getElementById('bot-panel');
  const msgs  = document.getElementById('bot-msgs');
  const input = document.getElementById('bot-input');
  const send  = document.getElementById('bot-send');

  showCaptchaUI();

  btn.addEventListener('click', () => {
    isOpen = !isOpen;
    panel.classList.toggle('open', isOpen);
    btn.textContent = isOpen ? '✕' : '🤖';
    if (isOpen) { msgs.scrollTop = msgs.scrollHeight; showCaptchaUI(); }
  });
  input.addEventListener('input', () => {
    input.style.height = 'auto';
    input.style.height = Math.min(input.scrollHeight, 80) + 'px';
  });
  input.addEventListener('keydown', e => {
    if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); sendMsg(); }
  });
  send.addEventListener('click', sendMsg);

  async function sendMsg() {
    if (!captchaVerified) { showCaptchaUI(); return; }
    const text = input.value.trim();
    if (!text || isLoading) return;

    addMsg('user', text);
    history.push({role: 'user', content: text});
    input.value = ''; input.style.height = 'auto';
    const typing = addTyping();
    isLoading = true; input.disabled = true; send.disabled = true;

    try {
      const res = await fetch(API, {
        method: 'POST', headers: HEADERS,
        body: JSON.stringify({
          message:       text,
          history:       history.slice(0, -1),
          source:        SOURCE,
          captcha_token: captchaToken || ''
        })
      });
      const data = await res.json();

      // If captcha failed server-side — reset and show captcha again
      if (data.captcha) {
        typing.remove();
        captchaVerified = false; captchaToken = null;
        sessionStorage.removeItem('bot_captcha');
        showCaptchaUI();
        if (window.turnstile) window.turnstile.reset();
        addMsg('bot', '⛔ Verification failed. Please verify again.');
        return;
      }

      typing.remove();
      const reply = data.response || 'Sorry, something went wrong.';
      if (data.ok) history.push({role: 'assistant', content: reply});
      addMsg('bot', reply);
    } catch(e) {
      typing.remove();
      addMsg('bot', '⚠️ Could not connect. Please try again.');
    } finally {
      isLoading = false; input.disabled = false; send.disabled = false; input.focus();
    }
  }

  function addMsg(role, text) {
    const d = document.createElement('div');
    d.className = 'bmsg ' + role;
    d.innerHTML = text
      .replace(/&/g,'&').replace(/'<').replace(/>/g,'>')
      .replace(/\*\*(.*?)\*\*/g,'<strong>$1</strong>')
      .replace(/\n/g,'<br>');
    msgs.appendChild(d); msgs.scrollTop = msgs.scrollHeight; return d;
  }
  function addTyping() {
    const d = document.createElement('div');
    d.className = 'btyping';
    d.innerHTML = '<div class="bdot"></div>'.repeat(3);
    msgs.appendChild(d); msgs.scrollTop = msgs.scrollHeight; return d;
  }
})();
</script>

Common problems

🔴
Captcha keeps asking every message
This means your backend is re-verifying every message. The app.py template above already fixes this with the verified_users set. Make sure you're using the exact template from Step 4 without changing the captcha logic.
🔴
No reply from bot at all
Check HF Space logs for errors. Most common causes: Groq API key not saved correctly in Secrets, or the API URL in the widget doesn't match your actual HF Space URL.
🔴
Knowledge not loading / wrong answers
Your GITHUB_BASE URL is wrong or the repo is private. Make sure the repo is set to Public and the URL format is exact: raw.githubusercontent.com/USERNAME/REPO/main
✅
Seeing 429 errors in logs
Normal — one model hit its rate limit and the bot automatically tried the next. The user still got a response. That's the fallback chain working correctly.

Add Basic Protection

The starter template already includes the essentials. Here's what's protected and what you can add as you grow.

Already included in the template

  • ✓
    Cloudflare Turnstile captchaVerified once per session, cached server-side. Blocks automated bots.
  • ✓
    Message length limitMessages over 500 characters are rejected before reaching the AI — zero tokens wasted.
  • ✓
    Daily limit per user150 messages/day per user. Resets at midnight. Stops one person from draining your quota.
  • ✓
    Cooldown between messages3-second minimum between messages. Stops rapid-fire spam.
  • ✓
    Model fallback chainIf one AI model fails or rate-limits, the next one is tried automatically.

Optional protections to add as your community grows

  • +
    NSFW/keyword filterBefore calling the AI, check the message for inappropriate keywords and reject instantly. Zero tokens wasted on bad content.
  • +
    Repetition detectionTrack recent messages per user. If the same message is sent 3+ times in 60 seconds, lock them briefly.
  • +
    Language filterIf your bot is English-only, check if the message contains non-ASCII letters and block before it reaches the AI.
  • +
    Strike systemTrack violations per user. First offense = short lock. Second = longer lock. Third = 5-minute lock. Deters persistent abusers.
💡
Core principle: Block as early as possible. Every check before the AI call costs zero tokens. A message that reaches the AI costs 500-2000 tokens. Put all your defenses in front of the AI call.

Maintain and Grow

Updating knowledge files

Go to your GitHub knowledge repo → click the .txt file → click the pencil icon → edit → commit. The bot picks up changes automatically within 10 minutes (the cache TTL). No restarts, no code changes.

Adding a new page with its own knowledge

  • 1
    Create a new .txt file in GitHub — e.g. faq.txt
  • 2
    On your new page change const SOURCE = 'faq' in the widget JavaScript
  • 3
    Done — that page's bot now loads faq.txt automatically

If AI models get decommissioned

Groq occasionally retires old model names. If you see decommissioned or model_not_found in your HF Space logs, visit console.groq.com/docs/models to see the current list and update the MODELS array in app.py.

Free tier limits to know

Groq
14,400 requests/day per model — resets midnight UTC. Multiple models in the fallback array multiplies your effective daily capacity.
Hugging Face
Free Spaces may sleep after 48h inactivity. They wake up automatically on the first request (takes ~30 seconds the first time).
GitHub
Unlimited public repos and files. No rate limits that affect knowledge file loading at normal usage.
Cloudflare Turnstile
Completely free forever. No request limits published for the free tier.
